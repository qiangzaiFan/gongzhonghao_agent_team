# System Inside Life Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `system_inside_life_agent/`, a long-term WeChat public account production workspace for “体制内清醒生活” with four rotating content columns and a hot-article deconstruction accelerator.

**Architecture:** Follow the existing single-track directory pattern used by `elder_healing_agent/`: local docs and Claude subagents define editorial behavior, Python scripts create prompt/run packets, and local gates check reference similarity, risk language, article shape, and AIGC detector status. Keep the first implementation self-contained in `system_inside_life_agent/` and reuse the shared detector through `astrology_content/ai_detector.py`.

**Tech Stack:** Python 3 standard library, `unittest`, Markdown prompt/spec files, JSON data files, existing shared AIGC detector script.

## Global Constraints

- New working directory is `system_inside_life_agent/`.
- Reference corpus root is `D:\自媒体\知识库\01-公众号文章\体制内公众号文章`.

- Content model is `四栏目矩阵 + 爆款拆解加速器`.
- Four columns are `新人避坑`, `基层现场`, `人情关系`, and `中年求稳`.
- Title modes are `original`, `inspired`, and `reference`.

- AIGC publish line is `human≥90%` and `ai≤10%`.
- Do not implement `--publish`, WeChat draft upload, credential loading, web UI, automatic link scraping, or image generation in the first version.
- Do not implement text-noise tricks for evading AIGC detection.
- Do not copy reference authors' body text, images, signature, personal cases, or identifiable body structure.
- Existing unrelated dirty files `.codegraph/.gitignore` and `.codex/hooks.json` must not be changed by this implementation.

- [ ] **Step 3: Implement corpus analyzer**

Create `system_inside_life_agent/scripts/analyze_reference_corpus.py` with these functions and constants:

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import json
import re
import sys
from collections import Counter, defaultdict
from dataclasses import asdict, dataclass
from pathlib import Path


BASE_DIR = Path(__file__).resolve().parents[1]
DEFAULT_REFERENCE_ROOT = Path(r"D:\自媒体\知识库\01-公众号文章\体制内公众号文章")
TITLE_OUT = BASE_DIR / "data" / "reference_titles.txt"
STATS_OUT = BASE_DIR / "data" / "style_stats.json"
REPORT_OUT = BASE_DIR / "references" / "corpus_report.md"
KEYWORDS = [
    "体制内", "乡镇", "基层", "领导", "提拔", "纪委", "问责", "编制",
    "事业编", "公务员", "县直", "上岸", "材料", "调动", "相亲",
    "婚姻", "女性", "情绪", "躺平", "退休", "饭局", "群众", "村干部",
    "副科", "人情", "留痕", "举报", "执法",
]


@dataclass(frozen=True)
class ArticleStat:
    author: str
    title: str
    path: str
    cjk: int
    paragraphs: int
    headings: int
    images: int
    title_pattern: str
    keywords: list[str]


def cjk_len(text: str) -> int:
    return len(re.findall(r"[\u4e00-\u9fff]", text))


def title_from_path(path: Path) -> str:
    stem = path.stem.strip()
    match = re.match(r"\d{4}-\d{2}-\d{2}\s+-\s+(.+)", stem)
    return match.group(1).strip() if match else stem


def title_pattern(title: str) -> str:
    if title.startswith("为什么"):
        return "为什么"
    if title.startswith("我在体制"):
        return "我在体制"
    if title.startswith("体制内"):
        return "体制内"
    if re.match(r"^[男女]，?\d{2}岁", title):
        return "问答身份"
    if title.startswith(("你知道", "你发现", "你注意")):
        return "你知道/发现/注意"
    return "其他"


def paragraph_count(text: str) -> int:
    return len([part for part in re.split(r"\n\s*\n", text) if part.strip()])


def analyze_markdown(path: Path, root: Path) -> ArticleStat:
    text = path.read_text(encoding="utf-8", errors="ignore")
    title = title_from_path(path)
    author = path.parent.name
    hits = [keyword for keyword in KEYWORDS if keyword in text or keyword in title]
    return ArticleStat(
        author=author,
        title=title,
        path=str(path.resolve()),
        cjk=cjk_len(text),
        paragraphs=paragraph_count(text),
        headings=len(re.findall(r"(?m)^#{1,4}\s+", text)),
        images=len(re.findall(r"!\[[^\]]*\]\([^)]+\)", text)),
        title_pattern=title_pattern(title),
        keywords=hits,
    )


def collect_articles(root: Path) -> list[ArticleStat]:
    if not root.exists():
        raise FileNotFoundError(f"参考目录不存在：{root}")
    return [analyze_markdown(path, root) for path in sorted(root.rglob("*.md")) if path.is_file()]


def unique_titles(stats: list[ArticleStat]) -> list[str]:
    seen: set[str] = set()
    result: list[str] = []
    for stat in stats:
        key = re.sub(r"\s+", "", stat.title)
        if key and key not in seen:
            seen.add(key)
            result.append(stat.title)
    return result


def build_style_stats(stats: list[ArticleStat]) -> dict[str, object]:
    by_author: dict[str, list[ArticleStat]] = defaultdict(list)
    for stat in stats:
        by_author[stat.author].append(stat)
    authors: dict[str, object] = {}
    for author, items in sorted(by_author.items()):
        keyword_counter = Counter(keyword for item in items for keyword in item.keywords)
        pattern_counter = Counter(item.title_pattern for item in items)
        authors[author] = {
            "count": len(items),
            "avg_cjk": round(sum(item.cjk for item in items) / max(len(items), 1)),
            "avg_paragraphs": round(sum(item.paragraphs for item in items) / max(len(items), 1), 1),
            "avg_headings": round(sum(item.headings for item in items) / max(len(items), 1), 1),
            "avg_images": round(sum(item.images for item in items) / max(len(items), 1), 1),
            "top_keywords": keyword_counter.most_common(12),
            "title_patterns": pattern_counter.most_common(),
        }
    return {"total_articles": len(stats), "authors": authors}


def build_report(stats: list[ArticleStat], style_stats: dict[str, object]) -> str:
    lines = ["# 体制内参考语料统计报告", ""]
    lines.append(f"- 总文章数：{style_stats['total_articles']}")
    lines.append("")
    lines.append("## 作者统计")
    for author, payload in style_stats["authors"].items():
        lines.append(f"### {author}")
        lines.append(f"- 文章数：{payload['count']}")
        lines.append(f"- 平均中文字符：{payload['avg_cjk']}")
        lines.append(f"- 平均段落数：{payload['avg_paragraphs']}")
        lines.append(f"- 平均小标题数：{payload['avg_headings']}")
        lines.append(f"- 平均图片数：{payload['avg_images']}")
        lines.append(f"- 高频关键词：{payload['top_keywords']}")
        lines.append(f"- 标题外壳：{payload['title_patterns']}")
        lines.append("")
    lines.append("## 原创边界")
    lines.append("- 本报告只用于抽象标题外壳、痛点、篇幅和结构统计。")
    lines.append("- 不复用参考博主正文、图片、署名、私人经历和可识别案例。")
    return "\n".join(lines) + "\n"


def write_outputs(stats: list[ArticleStat], base_dir: Path = BASE_DIR) -> None:
    data_dir = base_dir / "data"
    references_dir = base_dir / "references"
    data_dir.mkdir(parents=True, exist_ok=True)
    references_dir.mkdir(parents=True, exist_ok=True)
    titles = unique_titles(stats)
    style_stats = build_style_stats(stats)
    (data_dir / "reference_titles.txt").write_text("\n".join(titles) + "\n", encoding="utf-8")
    (data_dir / "style_stats.json").write_text(json.dumps(style_stats, ensure_ascii=False, indent=2), encoding="utf-8")
    (references_dir / "corpus_report.md").write_text(build_report(stats, style_stats), encoding="utf-8")


def main() -> int:
    parser = argparse.ArgumentParser(description="Analyze system-inside reference corpus.")
    parser.add_argument("--reference-root", type=Path, default=DEFAULT_REFERENCE_ROOT)
    args = parser.parse_args()
    try:
        stats = collect_articles(args.reference_root)
    except FileNotFoundError as exc:
        print(str(exc), file=sys.stderr)
        return 2
    write_outputs(stats)
    print(f"articles={len(stats)}")
    print(f"titles={TITLE_OUT}")
    print(f"stats={STATS_OUT}")
    print(f"report={REPORT_OUT}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

#!/usr/bin/env python3
from __future__ import annotations

import argparse
from pathlib import Path


def build_hotspot_prompt(source_path: Path) -> str:
    return f"""# 爆款拆解任务

请调用 `system-hotspot-analyst`，拆解下面参考文章。

来源文件：`{source_path}`

输出要求：

- 标题钩子
- 目标读者
- 痛点入口
- 冲突或反差
- 结构节奏
- 评论区可讨论点
- 可转化的新选题
- 不可复用内容清单

边界：

- 不生成正文。
- 不逐段改写来源文章。
- 不复用来源正文、案例、署名、图片和可识别段落结构。
"""


def main() -> int:
    parser = argparse.ArgumentParser(description="Create a hotspot deconstruction prompt.")
    parser.add_argument("source", type=Path)
    parser.add_argument("--out", type=Path)
    args = parser.parse_args()
    source = args.source.resolve()
    if not source.is_file():
        raise SystemExit(f"来源文件不存在：{source}")
    prompt = build_hotspot_prompt(source)
    if args.out:
        args.out.parent.mkdir(parents=True, exist_ok=True)
        args.out.write_text(prompt, encoding="utf-8")
        print(f"created={args.out}")
    else:
        print(prompt)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

from __future__ import annotations

import hashlib
import json
import tempfile
import unittest
from pathlib import Path
from unittest.mock import patch

import ai_detector


class SystemInsideAIDetectorTests(unittest.TestCase):
    def setUp(self) -> None:
        self.temp = tempfile.TemporaryDirectory()
        self.root = Path(self.temp.name)
        self.articles = self.root / "articles"
        self.reports = self.root / "reviews" / "auto"
        self.articles.mkdir(parents=True)

    def tearDown(self) -> None:
        self.temp.cleanup()

    def write_article(self, relative: str) -> Path:
        path = self.articles / relative
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text("---\ntitle: 测试\n---\n\n材料、会议通知、工作群都在眼前。\n", encoding="utf-8")
        return path

    def test_report_path_follows_article_relative_path(self) -> None:
        article = self.write_article("nested/draft.md")
        with patch.object(ai_detector, "ARTICLES_DIR", self.articles), patch.object(ai_detector, "REPORT_DIR", self.reports):
            self.assertEqual(ai_detector.default_report_path(article), self.reports / "nested" / "draft.json")

    def test_article_edit_invalidates_saved_report(self) -> None:
        article = self.write_article("draft.md")
        report = self.reports / "draft.json"
        report.parent.mkdir(parents=True)
        report.write_text(
            json.dumps(
                {
                    "article_sha256": hashlib.sha256(article.read_bytes()).hexdigest(),
                    "model": ai_detector.DEFAULT_MODEL,
                    "thresholds": {"human_min": 90.0, "ai_max": 10.0},
                    "ratios": {"human": 99.0, "suspected": 1.0, "ai": 0.0},
                    "passed": True,
                }
            ),
            encoding="utf-8",
        )
        self.assertIsNotNone(ai_detector.read_current_report(article, report, model=ai_detector.DEFAULT_MODEL, human_min=90.0, ai_max=10.0))
        article.write_text(article.read_text(encoding="utf-8") + "正文已修改。\n", encoding="utf-8")
        self.assertIsNone(ai_detector.read_current_report(article, report, model=ai_detector.DEFAULT_MODEL, human_min=90.0, ai_max=10.0))


if __name__ == "__main__":
    unittest.main()

- [ ] **Step 5: Implement batch quality script**

Create `system_inside_life_agent/scripts/batch_quality.py`:

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))

from quality_gate import analyze_article


ARTICLES_DIR = ROOT / "articles"
REVIEWS_DIR = ROOT / "reviews"


def collect_articles(articles_dir: Path = ARTICLES_DIR) -> list[Path]:
    if not articles_dir.exists():
        return []
    return sorted(path for path in articles_dir.rglob("*.md") if path.is_file())


def write_summary(reports: list[dict[str, object]], reviews_dir: Path = REVIEWS_DIR) -> None:
    reviews_dir.mkdir(parents=True, exist_ok=True)
    (reviews_dir / "quality_summary.json").write_text(json.dumps(reports, ensure_ascii=False, indent=2), encoding="utf-8")
    lines = ["# 体制内清醒生活批量质检", ""]
    for report in reports:
        lines.append(f"- {report['status']} score={report['score']} title={report['title']} path={report['path']}")
    (reviews_dir / "quality_summary.md").write_text("\n".join(lines) + "\n", encoding="utf-8")


def main() -> int:
    parser = argparse.ArgumentParser(description="Batch quality check system-inside drafts.")
    parser.add_argument("--articles-dir", type=Path, default=ARTICLES_DIR)
    parser.add_argument("--title-mode", choices=["original", "inspired", "reference"], default="inspired")
    args = parser.parse_args()
    reports = [analyze_article(path, title_mode=args.title_mode) for path in collect_articles(args.articles_dir)]
    write_summary(reports)
    failed = [report for report in reports if report["status"] != "passed"]
    print(f"checked={len(reports)} failed={len(failed)}")
    return 1 if failed else 0


if __name__ == "__main__":
    raise SystemExit(main())
```
